\section{Giới thiệu}
% ============================================================

% SLIDE 4
\begin{frame}{Vấn đề \& Động lực}
  \begin{itemize}
    \item CAPTCHA chữ bị méo, nhiễu, chồng lấn --- khó đọc bằng OCR truyền thống
    \item Thiếu dữ liệu huấn luyện chất lượng cho CAPTCHA tùy biến
    \item Nhiều hướng AI khác nhau, chưa so sánh trực tiếp trên cùng dataset
  \end{itemize}

  \vspace{0.4cm}
  \textbf{Giải pháp:}
  \[
    \text{Sinh dữ liệu} \rightarrow \text{Huấn luyện} \rightarrow \text{Demo so sánh 3 mô hình}
  \]
\end{frame}

% SLIDE 5
\begin{frame}{Mục tiêu dự án}
  \begin{enumerate}
    \item Sinh dataset CAPTCHA tổng hợp (5--8 ký tự, méo thực tế)
    \item Huấn luyện và đánh giá \textbf{YOLO11x}, \textbf{Donut}, \textbf{TrOCR}
    \item Xây giao diện Gradio so sánh inference trực quan
    \item Đo lường bằng CER, Exact Match, mAP (YOLO)
  \end{enumerate}
\end{frame}

% ============================================================
\section{Kiến trúc \& Dữ liệu}
% ============================================================

% SLIDE 6
\begin{frame}{Tổng quan ba mô hình}
  \begin{columns}[T]
    \column{0.33\textwidth}
    \textbf{YOLO11x}
    \begin{itemize}
      \item Detect từng ký tự
      \item 32 class
      \item Ghép chuỗi theo $x$
    \end{itemize}

    \column{0.33\textwidth}
    \textbf{Donut}
    \begin{itemize}
      \item Vision-Language
      \item Prompt \texttt{<s\_captcha>}
      \item Sinh text end-to-end
    \end{itemize}

    \column{0.33\textwidth}
    \textbf{TrOCR}
    \begin{itemize}
      \item OCR seq2seq
      \item \texttt{trocr-base-printed}
      \item Beam search
    \end{itemize}
  \end{columns}

  \vspace{0.5cm}
  \small
  Ba paradigm: \textbf{detection} (YOLO) vs \textbf{VLM} (Donut) vs \textbf{OCR end-to-end} (TrOCR)
\end{frame}

% SLIDE 7
\begin{frame}{Kiến trúc hệ thống}
  \begin{center}
    \small
    \texttt{config.py}
    $\downarrow$\\
    \texttt{gen\_image\_label.py} + \texttt{augment.py}
    $\rightarrow$ \texttt{generate\_dataset.py}
    $\downarrow$\\
    \texttt{captcha\_dataset/}
    $\downarrow$\\
    \texttt{YOLO.py} \;|\; \texttt{Donut\_train.py} \;|\; \texttt{TrOCR\_train.py}
    $\downarrow$\\
    \texttt{Weights/} $\rightarrow$ \texttt{app.py} (Gradio)
  \end{center}

  \vspace{0.3cm}
  Stack: Python 3.12, PyTorch, Transformers, Ultralytics, Albumentations, Gradio, WandB
\end{frame}

% SLIDE 8
\begin{frame}{Quy trình sinh dữ liệu}
  \begin{enumerate}
    \item \texttt{CaptchaGenerator}: render chữ, méo từng ký tự, xuất bbox YOLO
    \item \textbf{Train (9\,000 ảnh):} lưu ảnh gốc, augment on-the-fly khi train
    \item \textbf{Val (1\,000 ảnh):} augment cố định trước khi lưu
    \item \texttt{build\_yolo\_labels()}: nhãn 56 ký tự $\rightarrow$ 32 class (YOLO)
  \end{enumerate}
\end{frame}

% SLIDE 9
\begin{frame}{Cấu hình Dataset}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{Tham số chính}
    \begin{itemize}
      \item 10\,000 ảnh, train/val = 90/10
      \item Kích thước: $220 \times 60$ px
      \item 5--8 ký tự mỗi ảnh
      \item \texttt{VOCAB}: 56 ký tự
      \item \texttt{CHARSET}: 32 ký tự in hoa
    \end{itemize}

    \column{0.48\textwidth}
    \textbf{Cấu trúc thư mục}
    \begin{itemize}
      \item \texttt{train/images/}
      \item \texttt{train/OCR\_labels/}
      \item \texttt{train/labels/}
      \item \texttt{val/...} (tương tự)
    \end{itemize}
  \end{columns}
\end{frame}

% SLIDE 10
\begin{frame}{CaptchaGenerator --- Méo ảnh}
  \begin{itemize}
    \item Wave distortion (sin/cos mapping + \texttt{cv2.remap})
    \item Xoay $\pm 35°$, xiên, phóng/thu $[0.7, 1.2]$
    \item Chữ rỗng (hollow) --- xác suất 30\%
    \item Chồng lấn ký tự: \texttt{STEP\_MIN = -13}
    \item 10 font TTF, bbox YOLO chuẩn hóa từng ký tự
  \end{itemize}
\end{frame}

% SLIDE 11
\begin{frame}{Augmentation (CaptchaAugmenter)}
  \textbf{Custom transforms:}
  \begin{itemize}
    \item Nhiễu đường, chấm, gạch ngang, blur cục bộ, đảo màu vùng
  \end{itemize}

  \textbf{Pipeline Albumentations:}
  \begin{itemize}
    \item GridDistortion $\rightarrow$ OneOf (nhiễu) $\rightarrow$ OneOf (invert/noise) $\rightarrow$ ColorJitter
    \item Cập nhật bbox theo biến đổi (bbox-aware)
    \item Xem trước: \texttt{uv run sample.py}
  \end{itemize}
\end{frame}

% ============================================================
\section{Ba mô hình AI}
% ============================================================

% SLIDE 12
\begin{frame}{YOLO11x --- Object Detection}
  \textbf{Ý tưởng:} Mỗi ký tự = 1 bbox $\rightarrow$ sort theo $x$ $\rightarrow$ ghép chuỗi

  \vspace{0.3cm}
  \begin{itemize}
    \item Base: \texttt{yolo11x.pt}, 32 class (\texttt{CHARSET})
    \item Custom dataset/trainer + augment on-the-fly
    \item Tắt mosaic, flip, mixup mặc định của YOLO
    \item Train: batch 128, imgsz 320, 150 epochs
    \item Metric: mAP50, mAP50-95, CER, Exact Match
  \end{itemize}
\end{frame}

% SLIDE 13
\begin{frame}{Donut --- Vision-Language Model}
  \textbf{Ý tưởng:} Ảnh + prompt $\rightarrow$ decoder sinh chuỗi ký tự

  \vspace{0.2cm}
  \begin{itemize}
    \item Base: \texttt{naver-clova-ix/donut-base}
    \item Token: \texttt{<s\_captcha>TEXT</s\_captcha>}
    \item Kích thước ảnh: $128 \times 384$
    \item Patch \texttt{modeling\_donut\_swin.py} vào transformers
    \item Train: batch 64, lr $2 \times 10^{-5}$, 100 epochs, bf16
  \end{itemize}
\end{frame}

% SLIDE 14
\begin{frame}{TrOCR --- OCR Seq2Seq}
  \textbf{Ý tưởng:} Vision encoder + text decoder, đọc trực tiếp toàn ảnh

  \vspace{0.2cm}
  \begin{itemize}
    \item Base: \texttt{microsoft/trocr-base-printed}
    \item \texttt{Seq2SeqTrainer}, \texttt{predict\_with\_generate=True}
    \item Beam search: 4 beams, \texttt{max\_length=12}
    \item Không cần bbox, không cần prompt đặc biệt
    \item Best model theo CER (thay vì loss)
  \end{itemize}
\end{frame}

\begin{frame}{Metrics đánh giá}
  \begin{itemize}

    \item \textbf{Exact Match}: \% chuỗi đúng hoàn toàn

  \end{itemize}
